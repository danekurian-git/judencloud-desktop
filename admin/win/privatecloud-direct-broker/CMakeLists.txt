cmake_minimum_required(VERSION 3.24)

  message(FATAL_ERROR "This broker requires Windows and MSVC")

  dynamic_lease_policy.cpp broker_process_security.cpp broker_pipe_lifecycle.cpp)

target_compile_definitions(privateCloudDirectBroker PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set_property(TARGET privateCloudDirectBroker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(privateCloudDirectBroker PRIVATE /W4 /permissive- /guard:cf)
target_link_options(privateCloudDirectBroker PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
target_link_libraries(privateCloudDirectBroker PRIVATE fwpuclnt iphlpapi ws2_32 advapi32 ole32 shell32 rpcrt4 wintrust crypt32)
